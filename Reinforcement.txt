// R-1.7: Sum function
long sumSmallerThan(int n)
{
    long sum = 0;
    for (int i = 0; i < n; ++i) sum += i;
    return sum;
}

// R-1.8: isMultiple Function
bool isMultiple(long n, long m)
{
    return (m != 0) && (n % m == 0);
}
